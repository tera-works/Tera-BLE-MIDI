cmake_minimum_required(VERSION 3.16)
project(tera_tests C)
enable_testing()
add_executable(test_core test_core.c ../main/midi_parser.c ../main/ble_codec.c ../main/baud_detect.c)
target_include_directories(test_core PRIVATE ../main)
target_compile_options(test_core PRIVATE -Wall -Wextra -Werror -Wpedantic)
add_test(NAME midi_ble_baud COMMAND test_core)
add_executable(test_name test_name.c ../main/name_command.c ../main/midi_parser.c ../main/ble_codec.c)
target_include_directories(test_name PRIVATE ../main)
target_compile_options(test_name PRIVATE -Wall -Wextra -Werror -Wpedantic)
add_test(NAME midi_name COMMAND test_name)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_FOUND)
 add_library(baud_wave_bridge SHARED baud_wave_bridge.c ../main/midi_parser.c ../main/baud_detect.c)
 target_include_directories(baud_wave_bridge PRIVATE ../main)
 target_compile_options(baud_wave_bridge PRIVATE -Wall -Wextra -Werror)
 set_target_properties(baud_wave_bridge PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
 add_test(NAME baud_8n1_wave COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_baud_wave.py $<TARGET_FILE:baud_wave_bridge>)
endif()
